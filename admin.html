Step 2 — Track what people click

Add this to your site so you can see which items get added to the cart and how many people actually hit WhatsApp. Paste it below the snippet above.

<script> document.addEventListener('click', function (e) { var add = e.target.closest('[data-add-to-cart]'); if (add) track('add_to_cart', add.dataset.item || add.textContent.trim()); var wa = e.target.closest('a[href*="wa.me"]'); if (wa) track('whatsapp_click', location.pathname); var call = e.target.closest('a[href^="tel:"]'); if (call) track('call_click', location.pathname); }); function track(name, label) { var log = JSON.parse(localStorage.getItem('pp_events') || '[]'); log.push({ e: name, l: label, t: Date.now() }); localStorage.setItem('pp_events', JSON.stringify(log.slice(-500))); if (window.gtag) gtag('event', name, { label: label }); } </script>
