/* ============================================================
   Claude proxy — keeps your API key off the browser.

   NETLIFY:  save as  netlify/functions/claude.js
             endpoint becomes  /.netlify/functions/claude
   VERCEL:   save as  api/claude.js
             endpoint becomes  /api/claude

   Then set the environment variable ANTHROPIC_API_KEY in your
   Netlify/Vercel dashboard. Never put the key in your repo.

   Also set ADMIN_TOKEN to any password you choose, so random
   visitors can't run up your API bill. The admin page sends it
   as the x-admin-token header.
   ============================================================ */

export default async function handler(req, res) {
  // --- Vercel style (req, res) and Netlify style (req) both supported ---
  const isVercel = typeof res !== "undefined" && res && typeof res.status === "function";

  const send = (status, obj) =>
    isVercel
      ? res.status(status).json(obj)
      : new Response(JSON.stringify(obj), {
          status,
          headers: { "Content-Type": "application/json" }
        });

  if (req.method !== "POST") return send(405, { error: "POST only" });

  const headers = isVercel ? req.headers : Object.fromEntries(req.headers);
  const token = headers["x-admin-token"];

  if (process.env.ADMIN_TOKEN && token !== process.env.ADMIN_TOKEN) {
    return send(401, { error: "Not authorised" });
  }

  const body = isVercel ? req.body : await req.json();

  // Cap spend: refuse anything unusually large.
  if (JSON.stringify(body).length > 400000) {
    return send(413, { error: "Request too large" });
  }

  try {
    const upstream = await fetch("https://api.anthropic.com/v1/messages", {
      method: "POST",
      headers: {
        "Content-Type": "application/json",
        "x-api-key": process.env.ANTHROPIC_API_KEY,
        "anthropic-version": "2023-06-01"
      },
      body: JSON.stringify({
        model: body.model || "claude-sonnet-4-6",
        max_tokens: Math.min(body.max_tokens || 1000, 2000),
        system: body.system,
        messages: body.messages
      })
    });

    const data = await upstream.json();
    return send(upstream.status, data);
  } catch (err) {
    return send(500, { error: "Upstream failed", detail: String(err) });
  }
}

/* Netlify v2 functions want a named config export */
export const config = { path: "/api/claude" };
